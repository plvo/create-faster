Next.js installs `@mdx-js/loader`, `@mdx-js/react`, `@next/mdx` and `next-mdx-remote`. TanStack Start installs `@mdx-js/rollup`. Both install `remark-frontmatter` and `remark-mdx-frontmatter`.

## TanStack Start

Documents in `contents/` are compiled at build time by `@mdx-js/rollup` and loaded lazily with `import.meta.glob`, one chunk per document. YAML frontmatter is exported as `frontmatter`. The example route `/mdx/{-$slug}` renders `/mdx` as `home.mdx` and `/mdx/cool` as `cool.mdx`.

## Next.js

Documents in `contents/` are compiled at build time by `@next/mdx`, with `remark-frontmatter` and `remark-mdx-frontmatter` exporting the YAML frontmatter as `frontmatter`. The example route `/mdx/[[...slug]]` imports each document, so nothing reads the filesystem at request time and the route also runs on a Cloudflare Worker. It prerenders one page per document, `/mdx` for `home.mdx` and `/mdx/cool` for `cool.mdx`, and returns 404 for any other slug (`dynamicParams = false`). It also builds with `cloudflare-static`, where each document is exported as an HTML page. To add a document, create the file in `contents/` and add it to `documentLoaders` in `page.tsx`.

### mdx.ts

Frontmatter parsing utility:

```ts title="src/lib/mdx.ts"
export type MdxFrontmatter = {
  title: string;
  summary: string;
  publishedAt: string;
  imageUrl?: string;
};

export type ParsedFrontmatterReturn = {
  metadata: Partial<MdxFrontmatter>;
  content: string;
};

export function parseFrontmatter(fileContent: string): ParsedFrontmatterReturn {
  const frontmatterRegex = /---\s*([\s\S]*?)\s*---/;
  const match = frontmatterRegex.exec(fileContent);
  const frontMatterBlock = match![1];
  const content = fileContent.replace(frontmatterRegex, '').trim();
  const frontMatterLines = frontMatterBlock.trim().split('\n');
  const metadata: Partial<MdxFrontmatter> = {};

  frontMatterLines.forEach((line) => {
    const [key, ...valueArr] = line.split(': ');
    let value = valueArr.join(': ').trim();
    value = value.replace(/^['"](.*)['"]$/, '$1');
    metadata[key.trim() as keyof MdxFrontmatter] = value;
  });

  return { metadata, content };
}
```

### page.tsx

MDX route prerendered for every document, with metadata generation:

```tsx title="src/app/mdx/[[...slug]]/page.tsx"
import { notFound } from 'next/navigation';
import type { ComponentType } from 'react';
import type { MdxFrontmatter } from '@/lib/mdx';

const HOME_DOCUMENT = 'home';

type MdxDocument = {
  default: ComponentType;
  frontmatter: MdxFrontmatter;
};

const documentLoaders: Record<string, () => Promise<MdxDocument>> = {
  [HOME_DOCUMENT]: () => import('#/contents/home.mdx'),
  cool: () => import('#/contents/cool.mdx'),
};

interface PageProps {
  params: Promise<{ slug?: string[] }>;
}

export const dynamicParams = false;

export function generateStaticParams(): { slug: string[] }[] {
  return Object.keys(documentLoaders).map((name) => ({ slug: name === HOME_DOCUMENT ? [] : [name] }));
}

export const generateMetadata = async ({ params }: PageProps) => {
  const { slug } = await params;
  const { frontmatter } = await loadDocument(slug);
  return { title: frontmatter.title, description: frontmatter.summary };
};

export default async function Page({ params }: PageProps) {
  const { slug } = await params;
  const { default: Document } = await loadDocument(slug);

  return (
    <article className='max-w-162.5 mx-auto px-4 py-4 md:py-8'>
      <Document />
    </article>
  );
}

function loadDocument(slug: string[] = []): Promise<MdxDocument> {
  const isListed = generateStaticParams().some(
    (params) => params.slug.length === slug.length && params.slug.every((segment, index) => segment === slug[index]),
  );
  if (!isListed) notFound();

  return documentLoaders[slug[0] ?? HOME_DOCUMENT]();
}
```
